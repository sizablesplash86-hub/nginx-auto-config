void get_lan_ip(void) //void functions are in chapter 9 p. 183
{
    struct ifaddrs *ifaddr, *ifa; //pointers to structures, dynamic data structures in chapter 17 p. 416
    char ip[INET_ADDRSTRLEN];     //char arrays for strings in chapter 13 p. 280

    //getifaddrs allocates memory and builds linked list of network adapters (not in King book)
    if (getifaddrs(&ifaddr) == -1) //if statements covered in chapter 5.2 p. 74
    {
        perror("getifaddrs failed"); //error handling function in chapter 26.2 p. 696
        return;                      //return without value in void function chapter 9.2 p. 187
    }

    //for loops and pointers moving through a linked list, chapter 6.3 p. 110 & chapter 17.5 p. 428
    for (ifa = ifaddr; ifa != NULL; ifa = ifa->ifa_next) 
    {
        //skip empty pointers or addresses that aren't standard IPv4 (AF_INET)
        if (ifa->ifa_addr == NULL || ifa->ifa_addr->sa_family != AF_INET) 
        {
            continue; //continue statement chapter 6.4 p. 116
        }

        //filter out loopback, WireGuard, Tailscale, and Docker interfaces (strcmp & strncmp chapter 13.5 p. 300)
        if (strcmp(ifa->ifa_name, "lo") == 0 ||
            strncmp(ifa->ifa_name, "wg", 2) == 0 ||
            strncmp(ifa->ifa_name, "tailscale", 9) == 0 ||
            strncmp(ifa->ifa_name, "docker", 6) == 0) 
        {
            continue;
        }

        //struct casting (chapter 16 p. 377) from generic sockaddr to socket structure sockaddr_in
        struct sockaddr_in *pAddr = (struct sockaddr_in *)ifa->ifa_addr;
        
        //inet_ntop is a Linux conversion function, converts network binary IP to readable string
        inet_ntop(AF_INET, &pAddr->sin_addr, ip, sizeof(ip)); //sizeof operator chapter 7.6 p. 151

        printf("Primary LAN Interface (%s): %s\n", ifa->ifa_name, ip); //printf formatting chapter 3.1 p. 37
    }

    freeifaddrs(ifaddr); //frees dynamically allocated memory list, concept from free() chapter 17.4 p. 423
}

int main(void) //chapter 2.1 p. 9
{
    get_lan_ip();
    return 0; //chapter 2.1 p. 10
}
